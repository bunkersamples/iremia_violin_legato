on note
  if (search(%key_switches,$EVENT_NOTE) # -1)
    %p[$sp-1] := search(%key_switches,$EVENT_NOTE)
    call f__select_articulation
  end if
  disallow_group($ALL_GROUPS)
  if ($ADSR_swi_phase # 1)
    $lgt_phase_grp_ofs := 45
  else
    $lgt_phase_grp_ofs := 0
  end if
  if ($current_art # -1)
    $a := 0
    while ($a<5)
      if (get_control_par(%ui_channel_swi_mute[$a],$CONTROL_PAR_VALUE)=1)
        if ($solo_on_flag=0)
          $aa := 0
          while ($aa<3)
            allow_group(%art_start_group[$current_art]+$a+($aa*5)+$lgt_phase_grp_ofs)
            inc($aa)
          end while
        else
          if (%solo_channels[$a]=1)
            $aa := 0
            while ($aa<3)
              allow_group(%art_start_group[$current_art]+$a+($aa*5)+$lgt_phase_grp_ofs)
              inc($aa)
            end while
          end if
        end if
      end if
      inc($a)
    end while
  end if
  if ($swi_legato_poly=1 and ($current_art # -1) and (search(%key_switches,$EVENT_NOTE)=-1) and in_range($EVENT_NOTE,55,96))
    ignore_event($EVENT_ID)
    if (%lgt__parent[$lgt__new_idx]=0)
      $lgt__ref_time := $ENGINE_UPTIME
      $lgt__offset := 0
      %lgt__note[$lgt__new_idx] := $EVENT_NOTE
      %lgt__vel[$lgt__new_idx] := $EVENT_VELOCITY
      %lgt__parent[$lgt__new_idx] := $EVENT_ID
      %lgt__pedal[$lgt__new_idx] := 0
      %lgt__tune[$lgt__new_idx] := 0
      %lgt__child[$lgt__new_idx] := play_note($EVENT_NOTE,$EVENT_VELOCITY,$lgt__offset,0)
    else
      while ($lgt__busy=1)
        wait(100)
      end while
      if ($lgt__active=1)
        if ($lgt__fade_out_ticks>0)
          if (%lgt__child[1-$lgt__new_idx] # 0)
            note_off(%lgt__child[1-$lgt__new_idx])
          end if
          %lgt__pedal[1-$lgt__new_idx] := 0
          $lgt__fade_out_ticks := 0
        end if
        $lgt__fade_in_ticks := 0
        $lgt__seg2_ticks := 0
        $lgt__bend_ticks := 0
        $lgt__busy := 1
        while ($lgt__active=1)
          wait(100)
        end while
        $lgt__busy := 0
      end if
      $lgt__new_idx := 1-$lgt__new_idx
      $_elapsed := ($ENGINE_UPTIME-$lgt__ref_time)*1000
      if ($_elapsed<$SETTINGS_sli_legato_offset_min)
        $_elapsed := $SETTINGS_sli_legato_offset_min
      end if
      if ($_elapsed>$SETTINGS_sli_legato_offset_max)
        $_elapsed := $SETTINGS_sli_legato_offset_max
      end if
      $lgt__offset := $_elapsed
      $lgt__ref_time := $ENGINE_UPTIME
      %lgt__note[$lgt__new_idx] := $EVENT_NOTE
      %lgt__vel[$lgt__new_idx] := $EVENT_VELOCITY
      %lgt__parent[$lgt__new_idx] := $EVENT_ID
      %lgt__pedal[$lgt__new_idx] := 0
      %lgt__tune[$lgt__new_idx] := 0
      %lgt__child[$lgt__new_idx] := play_note($EVENT_NOTE,$EVENT_VELOCITY,$lgt__offset,0)
      $lgt__fade_out_ticks := $SETTINGS_sli_legato_time/100
      $lgt__fade_in_ticks := $lgt__fade_out_ticks*$SETTINGS_sli_legato_attack/100
      if ($SETTINGS_sli_legato_attack=100)
        $lgt__seg2_ticks := 0
        $lgt__atk1_time := $SETTINGS_sli_legato_time
      else
        $lgt__seg2_ticks := $lgt__fade_out_ticks-$lgt__fade_in_ticks
        $lgt__atk1_time := $SETTINGS_sli_legato_time*$SETTINGS_sli_legato_attack/$SETTINGS_sli_legato_node_vol
      end if
      if ($SETTINGS_swi_legato_bend_enable=1)
        $_interval := abs(%lgt__note[$lgt__new_idx]-%lgt__note[1-$lgt__new_idx])
      else
        $_interval := 0
      end if
      if ($_interval=0)
        $lgt__bend_ticks := 0
      else
        $lgt__bend_ticks := $SETTINGS_sli_legato_portamento_time/100
        $lgt__bend_amount := $SETTINGS_sli_legato_portamento_amount
        if ($SETTINGS_sli_legato_octave_x # 10 and ($_interval # 1))
          $lgt__bend_amount := $lgt__bend_amount+((($SETTINGS_sli_legato_octave_x-10)*($_interval-1)*$lgt__bend_amount+55)/110)
        end if
        if (%lgt__note[$lgt__new_idx]<%lgt__note[1-$lgt__new_idx])
          $lgt__bend_amount := -$lgt__bend_amount
        end if
        $lgt__bend_step := 1000*$lgt__bend_amount/$lgt__bend_ticks
      end if
      if ($SETTINGS_sli_legato_time=0)
        if (%lgt__child[1-$lgt__new_idx] # 0)
          note_off(%lgt__child[1-$lgt__new_idx])
        end if
        %lgt__pedal[1-$lgt__new_idx] := 0
        $lgt__fade_out_ticks := 0
      else
        fade_out(%lgt__child[1-$lgt__new_idx],$SETTINGS_sli_legato_time,0)
        fade_in(%lgt__child[$lgt__new_idx],$lgt__atk1_time)
        $lgt__active := 1
        while ($lgt__fade_out_ticks>0 or ($lgt__fade_in_ticks>0) or ($lgt__bend_ticks>0))
          dec($lgt__fade_out_ticks)
          dec($lgt__fade_in_ticks)
          dec($lgt__bend_ticks)
          if ($lgt__fade_out_ticks=0)
            if (%lgt__child[1-$lgt__new_idx] # 0)
              note_off(%lgt__child[1-$lgt__new_idx])
            end if
            %lgt__pedal[1-$lgt__new_idx] := 0
            $lgt__fade_out_ticks := 0
          end if
          if ($lgt__fade_in_ticks=0 and ($lgt__seg2_ticks>0))
            fade_in(%lgt__child[$lgt__new_idx],$lgt__seg2_ticks*100)
            $lgt__fade_in_ticks := $lgt__seg2_ticks
            $lgt__seg2_ticks := 0
          end if
          if ($lgt__bend_ticks>=0 and ($SETTINGS_swi_legato_bend_enable=1))
            $_old_tune := $lgt__bend_amount-(($lgt__bend_step*$lgt__bend_ticks+500)/1000)
            $_new_tune := -($lgt__bend_step*$lgt__bend_ticks+500)/1000
            change_tune(%lgt__child[1-$lgt__new_idx],$_old_tune-%lgt__tune[1-$lgt__new_idx],1)
            %lgt__tune[1-$lgt__new_idx] := $_old_tune
            change_tune(%lgt__child[$lgt__new_idx],$_new_tune-%lgt__tune[$lgt__new_idx],1)
            %lgt__tune[$lgt__new_idx] := $_new_tune
          end if
          wait(100)
        end while
        $lgt__active := 0
      end if
    end if
  end if
end on

on release
  if ($swi_legato_poly=1)
    if ($EVENT_ID=%lgt__parent[$lgt__new_idx])
      if ($lgt__pedal_down=0)
        while ($lgt__busy=1)
          wait(100)
        end while
        if ($lgt__active=1)
          if ($lgt__fade_out_ticks>0)
            if (%lgt__child[1-$lgt__new_idx] # 0)
              note_off(%lgt__child[1-$lgt__new_idx])
            end if
            %lgt__pedal[1-$lgt__new_idx] := 0
            $lgt__fade_out_ticks := 0
          end if
          $lgt__fade_in_ticks := 0
          $lgt__seg2_ticks := 0
          $lgt__bend_ticks := 0
          $lgt__busy := 1
          while ($lgt__active=1)
            wait(100)
          end while
          $lgt__busy := 0
        end if
        if (%lgt__child[0] # 0)
          note_off(%lgt__child[0])
        end if
        if (%lgt__child[1] # 0)
          note_off(%lgt__child[1])
        end if
        %lgt__child[0] := 0
        %lgt__child[1] := 0
        %lgt__parent[0] := 0
        %lgt__parent[1] := 0
        $lgt__new_idx := 0
      else
        %lgt__pedal[$lgt__new_idx] := 1
      end if
    else
      if ($EVENT_ID=%lgt__parent[1-$lgt__new_idx])
        if ($lgt__fade_out_ticks>0)
          if ($lgt__pedal_down=0)
            if (%lgt__child[1-$lgt__new_idx] # 0)
              note_off(%lgt__child[1-$lgt__new_idx])
            end if
            %lgt__pedal[1-$lgt__new_idx] := 0
            $lgt__fade_out_ticks := 0
          else
            %lgt__pedal[1-$lgt__new_idx] := 1
          end if
        end if
      end if
    end if
  end if
end on

          note_off(%lgt__child[1-$lgt__new_idx])
        end if
        %lgt__pedal[1-$lgt__new_idx] := 0
        $lgt__fade_out_ticks := 0
      end if
      if (%lgt__pedal[$lgt__new_idx]=1)
        %lgt__pedal[$lgt__new_idx] := 0
        while ($lgt__busy=1)
          wait(100)
        end while
        if ($lgt__active=1)
          if ($lgt__fade_out_ticks>0)

              note_off(%lgt__child[1-$lgt__new_idx])
            end if
            %lgt__pedal[1-$lgt__new_idx] := 0
            $lgt__fade_out_ticks := 0
          end if
          $lgt__fade_in_ticks := 0
          $lgt__seg2_ticks := 0
          $lgt__bend_ticks := 0
          $lgt__busy := 1
          while ($lgt__active=1)
            wait(100)
          end while

        end if

    end if
  end if
end on

on ui_control($SEND_FX_sli_reverb_lowcut)
  %reverb_cb_state[0] := $NI_CALLBACK_ID
  ~val := int_to_real($SEND_FX_sli_reverb_lowcut)*0.000001
  ~val := ~val*6.90775527898213+2.99573227355399
  ~val := pow(~NI_MATH_E,~val)
  $res := real_to_int(round(~val*10.0))
  if ($res<10000)
    set_text($SEND_FX_lbl_reverb_lowcut,$res/10 & "." & ($res mod 10) & " Hz")
  else
    set_text($SEND_FX_lbl_reverb_lowcut,$res/10000 & "." & (($res+500)/1000 mod 10) & "k" & " Hz")
  end if
  set_engine_par($ENGINE_PAR_IRC_FREQ_HIGHPASS_ER,$SEND_FX_sli_reverb_lowcut,-1,1,$NI_SEND_BUS)
  set_engine_par($ENGINE_PAR_IRC_FREQ_HIGHPASS_LR,$SEND_FX_sli_reverb_lowcut,-1,1,$NI_SEND_BUS)
  wait(2000000)
  if (%reverb_cb_state[0]=$NI_CALLBACK_ID)
    set_text($SEND_FX_lbl_reverb_lowcut,"LOW CUT")
  end if
end on
