function _twait
  %p[$sp-2] := search(%tstate__id,0)
  if (%p[$sp-2]=-1)
    pgs_set_key_val(TCM_EXCEPTION,$CURRENT_SCRIPT_SLOT,1)
  else
    %tstate__id[$tx] := $NI_CALLBACK_ID
    %tstate__fp[$tx] := $fp
    %tstate__sp[$tx] := $sp
    $tx := %p[$sp-2]
    %tstate__id[$tx] := -1
    $fp := %tstate__fs[$tx]+50
    %p[$fp-1] := %p[$sp-1]
    $sp := $fp
    wait(%p[$sp-1])
    %tstate__id[$tx] := 0
    $tx := search(%tstate__id,$NI_CALLBACK_ID)
    %tstate__id[$tx] := -1
    $fp := %tstate__fp[$tx]
    $sp := %tstate__sp[$tx]
  end if
end function

      set_key_color($_i4,$KEY_COLOR_INACTIVE)

    end while
  end if
  if ($_any_harmonics=1 and ($_any_normal=1))
    $_i4 := 78
    while ($_i4<=96)
      set_key_color($_i4,$KEY_COLOR_MINT)
      inc($_i4)
    end while
  else
    if ($_any_harmonics=1)
      $_i4 := 78
      while ($_i4<=96)
        set_key_color($_i4,$KEY_COLOR_GREEN)
        inc($_i4)
      end while
    else
      $_i4 := 78
      while ($_i4<=96)
        set_key_color($_i4,$KEY_COLOR_DEFAULT)
        inc($_i4)
      end while
    end if
  end if
  if ($_any_harmonics=1)
    $_i4 := 97
    while ($_i4<=108)
      set_key_color($_i4,$KEY_COLOR_GREEN)
      inc($_i4)
    end while
  else
    $_i4 := 97
    while ($_i4<=108)
      set_key_color($_i4,$KEY_COLOR_INACTIVE)
      inc($_i4)
    end while
  end if
  ~_half_pi := 1.5707963
  ~_x := ?xy_pad_xypad[0]
  ~_y := ?xy_pad_xypad[1]
  ~_blend := int_to_real($XY_pwr_blend)/1000.0
  ?_w_lin[0] := (1.0-~_x)*(1.0-~_y)
  ?_w_lin[1] := (1.0-~_x)*~_y
  ?_w_lin[2] := ~_x*~_y
  ?_w_lin[3] := ~_x*(1.0-~_y)
  ?_w_pow[0] := cos(~_x*~_half_pi)*cos(~_y*~_half_pi)
  ?_w_pow[1] := cos(~_x*~_half_pi)*sin(~_y*~_half_pi)
  ?_w_pow[2] := sin(~_x*~_half_pi)*sin(~_y*~_half_pi)
  ?_w_pow[3] := sin(~_x*~_half_pi)*cos(~_y*~_half_pi)
  $c := 0
  while ($c<4)
    ?_w[$c] := ?_w_lin[$c]+(~_blend*(?_w_pow[$c]-?_w_lin[$c]))
    select ($c)
      case 0
        $_trim_raw := $xy_pad_sli_trim_1
      case 1
        $_trim_raw := $xy_pad_sli_trim_2
      case 2
        $_trim_raw := $xy_pad_sli_trim_3
      case 3
        $_trim_raw := $xy_pad_sli_trim_4
    end select
    ~_trim_min_db := 0.0-~XY_TRIM_RANGE_DB
    ~_trim_pct := int_to_real($_trim_raw)/1000000.0
    ~_trim_range := 0.0-~_trim_min_db
    ~_trim_db := ~_trim_pct*~_trim_range
    ~_trim_db := ~_trim_db+~_trim_min_db
    ~_trim_lin := pow(10.0,~_trim_db/20.0)
    ?_w[$c] := ?_w[$c]*~_trim_lin
    $_cc_val := real_to_int((?_w[$c]-0.0)*127.0/1.0)
    if ($_cc_val>127)
      $_cc_val := 127
    else
      if ($_cc_val<0)
        $_cc_val := 0
      end if
    end if
    set_controller(104+$c,$_cc_val)
    inc($c)
  end while
